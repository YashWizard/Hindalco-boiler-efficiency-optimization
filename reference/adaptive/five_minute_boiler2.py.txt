class AdaptiveBoundsAndConstraintsManager:
    '\n    Manages adaptive bounds and constraints with stable 5-minute frequency updates for Boiler B2.\n    \n    This class automatically adjusts optimization bounds and constraint values\n    based on real-time plant data collected over 5-minute intervals for industrial stability.\n    '
    
    def __init__(self, buffer_margin=0.12):
        'Initialize the adaptive bounds manager with stable 5-minute updates.'
        self.buffer_margin = buffer_margin
        self.update_frequency = '5-minute'
        
        # Configure for stable 5-minute updates (more industrial appropriate)
        self.data_buffer_size = 300   # 5 minutes of data (300 seconds)
        self.update_interval = timedelta(minutes=5)  # Update every 5 minutes
        self.min_data_points = 60     # Need 60 seconds of data minimum for reliability
        self.max_bounds_change = 0.08 # Max 8% change per update for stability
        
        # Data storage for each parameter (5 minute rolling window)
        self.hourly_data = {i: deque(maxlen=self.data_buffer_size) for i in range(14)}
        
        # Initial bounds for Boiler B2 - CORRECTED parameter mapping
        self.current_bounds = [
            (2.1, 2.8),     # x[0] - Fuel-air ratio (B2 specific)
            (30, 100),      # x[1] - Wind box DP (B2 specific) - CORRECTED
            (150, 350),     # x[2] - PA fan primary air flow total - CORRECTED
            (10, 50),       # x[3] - Wind box air flow total - CORRECTED
            (120, 300),     # x[4] - Steam flow (B2 specific) - CORRECTED
            (290, 400),     # x[5] - ECO flue gas temp avg - CORRECTED
            (380, 420),     # x[6] - Superheater temp - CORRECTED
            (15.45, 90),    # x[7] - Coal flow (B2 specific) - CORRECTED
            (240, 350),     # x[8] - ESP duct gas temp avg - CORRECTED
            (150, 300),     # x[9] - FD water flow total - CORRECTED
            (110, 170),     # x[10] - FD water temp - CORRECTED
            (2, 6),         # x[11] - O2 - CORRECTED
            (2800, 3000),   # x[12] - Steam enthalpy - CORRECTED
            (500, 700)      # x[13] - FD water enthalpy - CORRECTED
        ]
        self.original_bounds = self.current_bounds.copy()
        
        # Adaptive constraint minimums for B2 - CORRECTED mapping
        self.current_constraint_mins = {
            'wind_box_dp_min': 30.0,        # x[1] - Wind box DP minimum
            'steam_flow_min': 120.0,        # x[4] - Steam flow minimum  
            'coal_flow_min': 15.45,         # x[7] - Coal flow minimum
            'air_fuel_ratio_min': 2.11,     # Fixed minimum air/fuel ratio for safety
        }
        
        self.last_update = datetime.now()
        print(f"Adaptive bounds configured for {self.update_frequency} updates (every 5 minutes) - Boiler B2")
        
    def add_real_time_data(self, parameter_values):
        'Add current parameter values to 5-minute rolling buffer.'
        for i, value in enumerate(parameter_values):
            if value is not None and isinstance(value, (int, float)) and not np.isnan(value):
                self.hourly_data[i].append(value)
    
    def calculate_adaptive_bounds_and_constraints(self):
        'Calculate new bounds based on past 5 minutes data with CORRECTED parameter mapping for B2.'
        new_bounds = []
        new_constraint_mins = self.current_constraint_mins.copy()
        
        for i in range(14):
            if len(self.hourly_data[i]) > self.min_data_points:
                data = list(self.hourly_data[i])
                
                # Use conservative percentiles for industrial stability
                p10 = np.percentile(data, 10)
                p90 = np.percentile(data, 90)
                range_buffer = (p90 - p10) * self.buffer_margin
                
                new_min = max(p10 - range_buffer, self.original_bounds[i][0] * 0.90)
                new_max = min(p90 + range_buffer, self.original_bounds[i][1] * 1.10)
                
                # Apply stability limits - prevent large jumps
                current_min, current_max = self.current_bounds[i]
                max_change_min = current_min * self.max_bounds_change
                max_change_max = current_max * self.max_bounds_change
                
                new_min = max(current_min - max_change_min, min(current_min + max_change_min, new_min))
                new_max = max(current_max - max_change_max, min(current_max + max_change_max, new_max))
                
                # Apply B2-specific constraint minimums - CORRECTED indices
                if i == 1:  # Wind box DP - CORRECTED
                    new_min = max(new_min, 30.0)
                    new_constraint_mins['wind_box_dp_min'] = new_min
                elif i == 4:  # Steam flow - CORRECTED
                    new_min = max(new_min, 120.0)
                    new_constraint_mins['steam_flow_min'] = new_min
                elif i == 7:  # Coal flow - CORRECTED
                    new_min = max(new_min, 15.45)
                    new_constraint_mins['coal_flow_min'] = new_min
                
                new_bounds.append((new_min, new_max))
            else:
                new_bounds.append(self.current_bounds[i])
                
        return new_bounds, new_constraint_mins
    
    def should_update_bounds(self):
        'Check if 5 minutes have passed since last update.'
        return (datetime.now() - self.last_update) >= self.update_interval
    
    def update_bounds_and_constraints_if_needed(self):
        'Update field_0001 and field_0002 field_0003 5 field_0011 field_0012 field_0013 field_0014.'
        if self.should_update_bounds():
            try:
                new_bounds, new_constraint_mins = self.calculate_adaptive_bounds_and_constraints()
                
                if self.validate_bounds_compatibility(new_bounds, new_constraint_mins):
                    self.current_bounds = new_bounds
                    self.current_constraint_mins = new_constraint_mins
                    self.last_update = datetime.now()
                    
                    # For 5-minute updates, keep more historical data for stability
                    for i in range(14):
                        if len(self.hourly_data[i]) > self.data_buffer_size * 0.9:
                            # Remove oldest 10% of data (more conservative)
                            for _ in range(int(len(self.hourly_data[i]) * 0.1)):
                                if self.hourly_data[i]:
                                    self.hourly_data[i].popleft()
                    
                    print(f"Adaptive bounds updated at {self.last_update.strftime('%H:%M:%S')} (5-minute frequency) - Boiler B2")
                    return True
                else:
                    print(f"Bounds validation failed at {datetime.now().strftime('%H:%M:%S')} - Boiler B2")
            except Exception as e:
                print(f"Bounds update error: {str(e)}")
        return False
    
    def validate_bounds_compatibility(self, new_bounds, new_constraint_mins):
        "Ensure new bounds don't create impossible constraint combinations for B2 - CORRECTED."
        try:
            for _ in range(5):  # More validation attempts
                test_values = [np.random.uniform(low, high) for (low, high) in new_bounds]
                
                # B2-specific constraint validation - CORRECTED indices
                if test_values[1] < new_constraint_mins['wind_box_dp_min']:  # Wind box DP
                    return False
                if test_values[4] < new_constraint_mins['steam_flow_min']:   # Steam flow
                    return False
                if test_values[7] < new_constraint_mins['coal_flow_min']:    # Coal flow
                    return False
                if (test_values[2] / test_values[7]) < new_constraint_mins['air_fuel_ratio_min']:  # Air/fuel ratio
                    return False
                    
            return True
        except:
            return False

    def objective(self, x):
        'Optimization objective function: maximize boiler efficiency.'
        input_df = pd.DataFrame([x], columns=self.model_features)

    def air_to_fuel_ratio(self, x):
        'Air-to-fuel ratio constraint function - CORRECTED indices.'
        constraint_mins = self.bounds_manager.current_constraint_mins
        return (x[2] / x[7]) - constraint_mins['air_fuel_ratio_min']  # PA fan air / coal flow

    def scc_constraint(self, x):
        'SCC constraint function: ensures SCC <= 0.202 - CORRECTED indices.'
        return self.max_scc - (x[7] / x[4])  # coal flow / steam flow

    def validate_solution(self, x):
        'Validate that optimization solution satisfies all constraints.'
        return all(c['fun'](x) >= -1e-6 for c in self.cons)

    def multi_start_optimization(self, n_runs=30):
        'Multi-start optimization using current adaptive bounds for B2 - reduced runs for stability.'
        # Use current adaptive bounds for optimization
        current_bounds = self.bounds_manager.current_bounds
        
        best_solution = None
        best_efficiency = -np.inf
        successful_runs = 0
        
        for i in range(n_runs):
            # Generate random initial guess within current adaptive bounds
            x0 = [np.random.uniform(low, high) for (low, high) in current_bounds]
            try:
                res = minimize(
                    self.objective,
                    x0,
                    method='SLSQP',
                    bounds=current_bounds,  # Use adaptive bounds
                    constraints=self.cons   # Use adaptive constraints
                )
                
                if res.success:
                    successful_runs += 1
                    print(f"[Run {i+1}] Success: Fun: {-res.fun:.4f}")
                    if (-res.fun > best_efficiency):
                        best_solution = res.x
                        best_efficiency = -res.fun
                else:
                    print(f"[Run {i+1}] Failed: {res.message}")
                    
            except Exception as e:
                print(f"[Run {i+1}] Optimization error: {e}")
                continue

        if best_solution is None:
            raise RuntimeError('All optimization attempts failed.')
            
        print(f"Optimization completed: {successful_runs}/{n_runs} successful runs")
        return best_solution, best_efficiency

    def safe_avg(self, values):
        'Calculate safe average of numeric values, handling None and non-numeric values.'
        numeric_values = [v for v in values if isinstance(v, (int, float))]
        if numeric_values:
            return sum(numeric_values) / len(numeric_values)
        else:
            return None

    def _get_batch_url(self, tag_url):
        'Extract batch API URL from individual tag URL.'
        parsed = urlparse(tag_url)
        return f"{parsed.scheme}://{parsed.netloc}/piwebapi/batch"

    def _batch_fetch(self, urls):
        'Fetch multiple PI tag values in a single batch request for efficiency.'
        if not urls:
            return

        batch_url = self._get_batch_url(urls[0])  # derive batch URL from first tag URL
        batch_payload = {
            f"req{i}": {
                'Method': 'GET',
                'Resource': url
            } for i, url in enumerate(urls)
        }

        try:
            response = requests.post(
                batch_url,
                json=batch_payload,
                auth=HTTPBasicAuth(self.username, self.password),
                verify=False
            )
            response.raise_for_status()
            batch_results = response.json()

            for i, url in enumerate(urls):
                result = batch_results.get(f"req{i}")
                value = None

                if result and 'Content' in result:
                    content = result['Content']
                    if isinstance(content, dict):
                        if 'Value' in content:
                            value = content['Value']
                        elif 'Value' in content.get('Value', {}):
                            value = content['Value']['Value']

                self._batch_cache[url] = value

        except Exception as e:
            print(f"Batch fetch error: {e}")
            for url in urls:
                self._batch_cache[url] = None

    def read_tag_value(self, url):
        'Read single PI tag value with batch processing optimization.'
        with self._batch_lock:
            now = time.time()
            self._batch_urls.add(url)

            if (now - self._last_batch_time) > self._batch_interval:
                urls_to_fetch = list(self._batch_urls)
                self._batch_urls.clear()
                self._last_batch_time = now
                self._batch_fetch(urls_to_fetch)

            if url in self._batch_cache:
                return self._batch_cache[url]

        # fallback to single fetch if not cached yet
        try:
            response = requests.get(url, auth=HTTPBasicAuth(self.username, self.password), verify=False)
            response.raise_for_status()
            data = response.json()

            value = data.get('Value')
            if isinstance(value, (int, float)):
                return value
            if isinstance(value, dict):
                if 'Value' in value:
                    return value['Value']
                else:
                    return None

            return None

        except Exception as e:
            print(f"Error fetching from {url}: {e}")
            return None
