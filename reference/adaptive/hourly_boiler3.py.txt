class AdaptiveBoundsAndConstraintsManager:
    '\n    Manages adaptive bounds and constraints for SCC optimization.\n    \n    This class automatically adjusts optimization bounds and constraint values\n    based on real-time plant data collected over hourly intervals. It uses\n    statistical analysis of actual operating parameters to create realistic\n    bounds that reflect current plant conditions.\n    '
    
    def __init__(self, buffer_margin=0.15):
        '\n        Initialize the adaptive bounds manager.\n        \n        Args:\n            buffer_margin (float): Safety margin for bounds expansion (default: 15%)\n        '
        self.buffer_margin = buffer_margin
        
        # Store hourly data for each of the 14 parameters
        # Each deque has max length 3600 (assuming 1-second intervals for 1 hour)
        self.hourly_data = {i: deque(maxlen=3600) for i in range(14)}
        
        # Initial bounds - these will be updated based on real plant data
        # Updated to ensure constraint compatibility
        self.current_bounds = [
            (4.5, 5.5),    # x[0] - Fuel-air ratio
            (190, 300),    # x[1] - Total air flow (increased to satisfy air/fuel constraint)
            (38, 100),     # x[2] - Wind box DP (increased to meet constraint minimum)
            (10, 50),      # x[3] - PA fan primary air flow
            (10, 50),      # x[4] - Wind box air flow
            (0.5, 1.5),    # x[5] - Secondary to primary ratio
            (130, 300),    # x[6] - Steam flow (increased to meet constraint minimum)
            (380, 420),    # x[7] - Superheater temp
            (290, 350),    # x[8] - ECO gas temp
            (35, 90),      # x[9] - Coal flow (increased to meet constraint minimum)
            (240, 300),    # x[10] - ESP duct gas temp
            (150, 250),    # x[11] - FD water flow
            (110, 170),    # x[12] - FD water temp
            (2, 6)         # x[13] - O2
        ]
        
        # Keep original bounds as reference for safety limits
        self.original_bounds = self.current_bounds.copy()
        
        # Adaptive constraint minimum values (these update with bounds)
        self.current_constraint_mins = {
            'wind_box_dp_min': 38.0,        # Minimum wind box DP for operation
            'steam_flow_min': 130.0,        # Minimum steam flow for stable operation
            'coal_flow_min': 35.0,          # Minimum coal flow for efficient combustion
            'air_fuel_ratio_min': 2.11,     # Fixed minimum air/fuel ratio for safety
        }
        
        # Track last update time for hourly intervals
        self.last_update = datetime.now()
        
    def add_real_time_data(self, parameter_values):
        '\n        Add current parameter values to rolling hourly buffers.\n        \n        Args:\n            parameter_values (list): List of 14 current parameter values from PI tags\n        '
        for i, value in enumerate(parameter_values):
            # Only add valid numeric values (exclude None and NaN)
            if value is not None and isinstance(value, (int, float)) and not np.isnan(value):
                self.hourly_data[i].append(value)
    
    def calculate_adaptive_bounds_and_constraints(self):
        "\n        Calculate new bounds and constraint minimums based on past hour's data.\n        \n        Uses statistical analysis (10th-90th percentiles) to determine realistic\n        operating ranges while maintaining safety margins and constraint compatibility.\n        \n        Returns:\n            tuple: (new_bounds, new_constraint_mins)\n        "
        new_bounds = []
        new_constraint_mins = self.current_constraint_mins.copy()
        
        # Process each parameter
        for i in range(14):
            # Check if we have sufficient data for reliable statistics
            if len(self.hourly_data[i]) > 20:
                data = list(self.hourly_data[i])
                
                # Use robust percentiles to avoid outliers
                p10 = np.percentile(data, 10)  # 10th percentile as lower bound
                p90 = np.percentile(data, 90)  # 90th percentile as upper bound
                
                # Add buffer margin for optimization exploration
                range_buffer = (p90 - p10) * self.buffer_margin
                new_min = max(p10 - range_buffer, self.original_bounds[i][0] * 0.85)
                new_max = min(p90 + range_buffer, self.original_bounds[i][1] * 1.15)
                
                # Apply parameter-specific constraint enforcement
                if i == 1:  # Air flow - ensure minimum for air/fuel ratio
                    new_min = max(new_min, 190.0)
                elif i == 2:  # Wind box DP - operational minimum
                    new_min = max(new_min, 38.0)
                    new_constraint_mins['wind_box_dp_min'] = new_min
                elif i == 6:  # Steam flow - operational minimum
                    new_min = max(new_min, 130.0)
                    new_constraint_mins['steam_flow_min'] = new_min
                elif i == 9:  # Coal flow - operational minimum
                    new_min = max(new_min, 35.0)
                    new_constraint_mins['coal_flow_min'] = new_min
                
                new_bounds.append((new_min, new_max))
            else:
                # Insufficient data - keep current bounds
                new_bounds.append(self.current_bounds[i])
                
        return new_bounds, new_constraint_mins
    
    def should_update_bounds(self):
        "\n        field_0089 field_0090 field_0091'field_0092 field_0088 field_0012 field_0093 field_0001 update.\n        \n        field_0094:\n            field_0095: field_0096 field_0090 field_0097 field_0098 field_0099 field_0100 field_0101 field_0102 update\n        "
        return (datetime.now() - self.last_update) >= timedelta(hours=1)
    
    def update_bounds_and_constraints_if_needed(self):
        '\n        Update field_0001 and field_0002 field_0090 field_0093 field_0103 field_0099 field_0100.\n        \n        field_0094:\n            field_0095: field_0096 field_0090 field_0001 field_0104 field_0105, field_0106 field_0107\n        '
        if self.should_update_bounds():
            try:
                # Calculate new adaptive bounds and constraints
                new_bounds, new_constraint_mins = self.calculate_adaptive_bounds_and_constraints()
                
                # Validate that new bounds don't create impossible constraint combinations
                if self.validate_bounds_compatibility(new_bounds, new_constraint_mins):
                    # Apply new bounds and constraints
                    self.current_bounds = new_bounds
                    self.current_constraint_mins = new_constraint_mins
                    self.last_update = datetime.now()
                    
                    # Clear data buffers for next hour
                    for i in range(14):
                        self.hourly_data[i].clear()
                        
                    print(f"Adaptive bounds updated at {self.last_update.strftime('%H:%M:%S')}")
                    return True
                else:
                    print('Bounds validation failed, keeping current values')
            except Exception as e:
                print(f"Bounds update error: {str(e)}")
        return False
    
    def validate_bounds_compatibility(self, new_bounds, new_constraint_mins):
        "\n        Validate that new bounds don't create impossible constraint combinations.\n        \n        Args:\n            new_bounds (list): Proposed new bounds\n            new_constraint_mins (dict): Proposed new constraint minimums\n            \n        Returns:\n            bool: True if bounds are compatible with constraints\n        "
        try:
            # Test several random samples from new bounds
            for _ in range(3):
                test_values = [np.random.uniform(low, high) for (low, high) in new_bounds]
                
                # Check critical operational constraints
                if test_values[2] < new_constraint_mins['wind_box_dp_min']:
                    return False
                if test_values[6] < new_constraint_mins['steam_flow_min']:
                    return False
                if test_values[9] < new_constraint_mins['coal_flow_min']:
                    return False
                if (test_values[1] / test_values[9]) < new_constraint_mins['air_fuel_ratio_min']:
                    return False
                    
            return True
        except:
            return False

    def update_constraints(self):
        '\n        Update field_0005 field_0002 field_0006 field_0007 field_0008 field_0009 field_0010.\n        \n        field_0108 field_0109 field_0110 field_0111 field_0112 field_0113 field_0114 field_0115 field_0008 field_0009\n        field_0116 values from field_0115 field_0001 field_0117, field_0118 field_0002 field_0119\n        with field_0120 field_0121 field_0122.\n        '
        constraint_mins = self.bounds_manager.current_constraint_mins
        
        # Define adaptive constraints for optimization
        self.cons = [
            # Air-to-fuel ratio constraint (safety requirement)
            {'type': 'ineq', 'fun': self.air_to_fuel_ratio},
            
            # Coal flow minimum (adaptive based on recent data)
            {'type': 'ineq', 'fun': lambda x: x[9] - constraint_mins['coal_flow_min']},
            
            # Steam flow minimum (adaptive based on recent data)
            {'type': 'ineq', 'fun': lambda x: x[6] - constraint_mins['steam_flow_min']},
            
            # Wind box DP minimum (adaptive based on recent data)
            {'type': 'ineq', 'fun': lambda x: x[2] - constraint_mins['wind_box_dp_min']},
            
            # SCC constraint (fixed maximum for regulatory compliance)
            {'type': 'ineq', 'fun': self.scc_constraint},
        ]

    def objective(self, x):
        '\n        Optimization objective function: maximize boiler efficiency.\n        \n        Args:\n            x (list): Parameter values for optimization\n            \n        Returns:\n            float: Negative efficiency (for minimization)\n        '
        input_df = pd.DataFrame([x], columns=self.model_features)
        return -self.model.predict(input_df)[0]

    def air_to_fuel_ratio(self, x):
        '\n        Air-to-fuel ratio constraint function.\n        \n        Args:\n            x (list): Parameter values\n            \n        Returns:\n            float: Constraint value (must be >= 0 for feasibility)\n        '
        constraint_mins = self.bounds_manager.current_constraint_mins
        return (x[1] / x[9]) - constraint_mins['air_fuel_ratio_min']

    def scc_constraint(self, x):
        '\n        SCC constraint function: ensures SCC <= 0.202.\n        \n        Args:\n            x (list): Parameter values\n            \n        Returns:\n            float: Constraint value (must be >= 0 for feasibility)\n        '
        return self.max_scc - (x[9] / x[6])

    def validate_solution(self, x):
        '\n        Validate that optimization solution satisfies all constraints.\n        \n        Args:\n            x (list): Parameter values to validate\n            \n        Returns:\n            bool: True if all constraints are satisfied\n        '
        return all(c['fun'](x) >= -1e-6 for c in self.cons)
        
    def multi_start_optimization(self, n_runs=50):
        '\n        Multi-start optimization to find global optimum.\n        \n        Runs the optimizer from multiple random starting points within\n        adaptive bounds to increase chance of finding global optimum\n        and avoid local minima.\n        \n        Args:\n            n_runs (int): Number of optimization attempts\n            \n        Returns:\n            tuple: (best_solution, best_efficiency)\n        '
        # Use current adaptive bounds for optimization
        current_bounds = self.bounds_manager.current_bounds
        
        best_solution = None
        best_efficiency = -np.inf
        
        # Run optimization from multiple random starting points
        for i in range(n_runs):
            # Generate random initial guess within adaptive bounds
            x0 = [np.random.uniform(low, high) for (low, high) in current_bounds]
            
            try:
                # Run local optimization with SLSQP method
                res = minimize(
                    self.objective,           # Maximize efficiency
                    x0,                      # Starting point
                    method='SLSQP',          # Sequential Quadratic Programming
                    bounds=current_bounds,    # Parameter bounds
                    constraints=self.cons     # Operational constraints
                )
                
                # Log optimization attempt results
                print(f"[Run {i+1}] Success: {res.success}, Fun: {-res.fun if res.success else 'N/A'}, Message: {res.message}")
                
                # Update best solution if this attempt is better
                if res.success and (-res.fun > best_efficiency):
                    best_solution = res.x
                    best_efficiency = -res.fun
                    
            except Exception as e:
                print(f"[Run {i+1}] Optimization error: {e}")
                continue
    
        # Check if any optimization succeeded
        if best_solution is None:
            raise RuntimeError('All optimization attempts failed.')
            
        return best_solution, best_efficiency

    def safe_avg(self, values):
        '\n        Calculate safe average of numeric values, handling None and non-numeric values.\n        \n        Args:\n            values (list): List of values to average\n            \n        Returns:\n            float or None: Average of numeric values, or None if no valid values\n        '
        numeric_values = [v for v in values if isinstance(v, (int, float))]
        if numeric_values:
            return sum(numeric_values) / len(numeric_values)
        else:
            return None

    def _get_batch_url(self, tag_url):
        '\n        Extract batch API URL from individual tag URL.\n        \n        Args:\n            tag_url (str): Individual PI tag URL\n            \n        Returns:\n            str: Batch API URL\n        '
        parsed = urlparse(tag_url)
        return f"{parsed.scheme}://{parsed.netloc}/piwebapi/batch"

    def _batch_fetch(self, urls):
        '\n        Fetch multiple PI tag values in a single batch request for efficiency.\n        \n        Args:\n            urls (list): List of PI tag URLs to fetch\n        '
        if not urls:
            return

        batch_url = self._get_batch_url(urls[0])
        
        # Create batch payload with multiple GET requests
        batch_payload = {
            f"req{i}": {
                'Method': 'GET',
                'Resource': url
            } for i, url in enumerate(urls)
        }

        try:
            # Execute batch request
            response = requests.post(
                batch_url,
                json=batch_payload,
                auth=HTTPBasicAuth(self.username, self.password),
                verify=False
            )
            response.raise_for_status()
            batch_results = response.json()

            # Process batch results and cache values
            for i, url in enumerate(urls):
                result = batch_results.get(f"req{i}")
                value = None

                if result and 'Content' in result:
                    content = result['Content']
                    if isinstance(content, dict):
                        if 'Value' in content:
                            value = content['Value']
                        elif 'Value' in content.get('Value', {}):
                            value = content['Value']['Value']

                self._batch_cache[url] = value

        except Exception as e:
            print(f"Batch fetch error: {e}")
            # Set None for all URLs on batch failure
            for url in urls:
                self._batch_cache[url] = None

    def read_tag_value(self, url):
        '\n        Read single PI tag value with batch processing optimization.\n        \n        Args:\n            url (str): PI tag URL\n            \n        Returns:\n            float or None: Tag value, or None if read failed\n        '
        # Use batch processing for efficiency
        with self._batch_lock:
            now = time.time()
            self._batch_urls.add(url)

            # Execute batch if interval elapsed
            if (now - self._last_batch_time) > self._batch_interval:
                urls_to_fetch = list(self._batch_urls)
                self._batch_urls.clear()
                self._last_batch_time = now
                self._batch_fetch(urls_to_fetch)

            # Return cached value if available
            if url in self._batch_cache:
                return self._batch_cache[url]

        # Fallback to individual fetch if not cached
        try:
            response = requests.get(url, auth=HTTPBasicAuth(self.username, self.password), verify=False)
            response.raise_for_status()
            data = response.json()

            value = data.get('Value')
            if isinstance(value, (int, float)):
                return value
            if isinstance(value, dict):
                if 'Value' in value:
                    return value['Value']

            return None

        except Exception as e:
            print(f"Error fetching from {url}: {e}")
            return None
