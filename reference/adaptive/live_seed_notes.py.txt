class AdaptiveBoundsAndConstraintsManager:
   
    def __init__(self, buffer_margin=0.15):
        self.buffer_margin = buffer_margin
        self.update_frequency = 'minute'
        self.data_buffer_size = 180   # 3 min  seconds of data
        self.update_interval = timedelta(minutes=1)
        self.min_data_points = 10     # Need 10 seconds of data minimum
        self.recent_data = {i: deque(maxlen=self.data_buffer_size) for i in range(14)}
        self.current_bounds = [(0, 1e6)] * 14  # Start with wide bounds, will be learned
        self.current_constraint_mins = {}
        self.last_update = datetime.now()
        print(f"Adaptive bounds manager initialized (no hardcoded bounds).")

    def add_real_time_data(self, parameter_values):
        for i, value in enumerate(parameter_values):
            if value is not None and isinstance(value, (int, float)) and not np.isnan(value):
                self.recent_data[i].append(value)

    def calculate_adaptive_bounds_and_constraints(self):
        new_bounds = []
        mins = {}
        for i in range(14):
            data = list(self.recent_data[i])
            if len(data) > self.min_data_points:
                p5 = np.percentile(data, 5)
                p95 = np.percentile(data, 95)
                range_buffer = (p95 - p5) * self.buffer_margin
                new_min = max(0, p5 - range_buffer)
                new_max = p95 + range_buffer
                new_bounds.append((new_min, new_max))
                # For constraints, use 10th percentile as a robust operational minimum
                if i in [1, 2, 6, 9, 10]:
                    mins[i] = np.percentile(data, 10)
            else:
                new_bounds.append((0, 1e6))
        constraint_map = {1: 'total_air_flow_min', 2: 'wind_box_dp_min', 6: 'steam_flow_min', 9: 'coal_flow_min', 10: 'aph_outlet_min'}
        new_constraint_mins = {constraint_map[i]: mins[i] for i in mins if i in constraint_map}
        # Air/fuel ratio constraint: use 10th percentile of air/fuel over last minute
        air_vals = list(self.recent_data[1])
        coal_vals = list(self.recent_data[9])
        if len(air_vals) > self.min_data_points and len(coal_vals) > self.min_data_points:
            ratios = [a / c for a, c in zip(air_vals, coal_vals) if c > 0]
            if ratios:
                new_constraint_mins['air_fuel_ratio_min'] = np.percentile(ratios, 10)
            else:
                new_constraint_mins['air_fuel_ratio_min'] = 2.0
        else:
            new_constraint_mins['air_fuel_ratio_min'] = 2.0
        return new_bounds, new_constraint_mins

    def should_update_bounds(self):
        return (datetime.now() - self.last_update) >= self.update_interval

    def update_bounds_and_constraints_if_needed(self):
        if self.should_update_bounds():
            try:
                new_bounds, new_constraint_mins = self.calculate_adaptive_bounds_and_constraints()
                if self.validate_bounds_compatibility(new_bounds, new_constraint_mins):
                    self.current_bounds = new_bounds
                    self.current_constraint_mins = new_constraint_mins
                    self.last_update = datetime.now()
                    print(f"Adaptive bounds updated at {self.last_update.strftime('%H:%M:%S')} (learned from 1-min data)")
                    return True
                else:
                    print(f"Bounds validation failed at {datetime.now().strftime('%H:%M:%S')}")
            except Exception as e:
                print(f"Bounds update error: {str(e)}")
        return False

                "test_values = [np.random.uniform(low, high) for (low, high) in new_bounds]\n                if test_values[1] < new_constraint_mins.get('total_air_flow_min', 0):\n                    return False\n                if test_values[2] < new_constraint_mins.get('wind_box_dp_min', 0):\n                    return False\n                if test_values[6] < new_constraint_mins.get('steam_flow_min', 0):\n                    return False\n                if test_values[9] < new_constraint_mins.get('coal_flow_min', 0):\n                    return False\n                if test_values[10] < new_constraint_mins.get('aph_outlet_min', 0):\n                    return False\n                if (test_values[1] / test_values[9]) < new_constraint_mins.get('air_fuel_ratio_min', 2.0):\n                    return False"

    def update_constraints(self):
        cm = self.bounds_manager.current_constraint_mins
        self.cons = [
            {'type': 'ineq', 'fun': lambda x: x[1] - cm.get('total_air_flow_min', 0)},
            {'type': 'ineq', 'fun': self.air_to_fuel_ratio},
            {'type': 'ineq', 'fun': lambda x: x[9] - cm.get('coal_flow_min', 0)},
            {'type': 'ineq', 'fun': lambda x: x[6] - cm.get('steam_flow_min', 0)},
            {'type': 'ineq', 'fun': lambda x: x[10] - cm.get('aph_outlet_min', 0)},
            {'type': 'ineq', 'fun': lambda x: x[2] - cm.get('wind_box_dp_min', 0)},
            {'type': 'ineq', 'fun': self.scc_constraint},
        ]

    def objective(self, x):
        input_df = pd.DataFrame([x], columns=self.model_features)
        return -self.model.predict(input_df)[0]

    def air_to_fuel_ratio(self, x):
        cm = self.bounds_manager.current_constraint_mins
        return (x[1] / x[9]) - cm.get('air_fuel_ratio_min', 2.0)

    def scc_constraint(self, x):
        return self.max_scc - (x[9] / x[6])

    def validate_solution(self, x):
        return all(c['fun'](x) >= -1e-6 for c in self.cons)

    def multi_start_optimization(self, live_inputs, n_runs=30):
        current_bounds = self.bounds_manager.current_bounds
        best_solution = None
        best_efficiency = -np.inf
        for i in range(n_runs):
            x0 = live_inputs.copy()
            for j in range(len(x0)):
                if x0[j] is not None:
                    perturbation = x0[j] * 0.03 * (np.random.random() - 0.5)
                    x0[j] += perturbation
                    x0[j] = max(current_bounds[j][0], min(current_bounds[j][1], x0[j]))
            try:
                res = minimize(
                    self.objective,
                    x0,
                    method='SLSQP',
                    bounds=current_bounds,
                    constraints=self.cons
                )
                if res.success and (-res.fun > best_efficiency):
                    best_solution = res.x
                    best_efficiency = -res.fun
            except Exception:
                continue
        if best_solution is None:
            return live_inputs, 0
        return best_solution, best_efficiency

    def read_tag_value(self, url):
        try:
            response = requests.get(url, auth=HTTPBasicAuth(self.username, self.password), verify=False)
            response.raise_for_status()
            data = response.json()
            value = data.get('Value')
            if isinstance(value, (int, float)):
                return value
            if isinstance(value, dict):
                if 'Value' in value:
                    return value['Value']
                else:
                    return None
            return None
        except Exception as e:
            print(f"Error fetching from {url}: {e}")
            return None
