class AdaptiveBoundsAndConstraintsManager:
    '\n    Manages adaptive bounds and constraints with 1-minute frequency updates for Boiler B2.\n    \n    This class automatically adjusts optimization bounds and constraint values\n    based on real-time plant data collected over 1-minute intervals.\n    '
    
    def __init__(self, buffer_margin=0.15):
        'Initialize the adaptive bounds manager for 1-minute updates.'
        self.buffer_margin = buffer_margin
        self.update_frequency = 'minute'
        
        # Configure for 1-minute updates
        self.data_buffer_size = 60    # 60 seconds of data
        self.update_interval = timedelta(minutes=1)
        self.min_data_points = 10     # Need 10 seconds of data minimum
        
        # Data storage for each parameter (1 minute rolling window)
        self.hourly_data = {i: deque(maxlen=self.data_buffer_size) for i in range(14)}
        
        # Initial bounds for Boiler B2 (adapted from original B2 bounds)
        self.current_bounds = [
            (2.1, 2.8),    # x[0] - Fuel-air ratio (B2 specific)
            (150, 300),    # x[1] - Total air flow (increased to satisfy air/fuel constraint)
            (30, 100),     # x[2] - Wind box DP (B2 specific)
            (10, 50),      # x[3] - PA fan primary air flow
            (10, 50),      # x[4] - Wind box air flow
            (0.5, 1.5),    # x[5] - Secondary to primary ratio
            (120, 300),    # x[6] - Steam flow (B2 specific)
            (380, 420),    # x[7] - Superheater temp
            (290, 350),    # x[8] - ECO gas temp
            (15.45, 90),   # x[9] - Coal flow (B2 specific)
            (240, 300),    # x[10] - ESP duct gas temp
            (150, 250),    # x[11] - FD water flow
            (110, 170),    # x[12] - FD water temp
            (2, 6)         # x[13] - O2
        ]
        self.original_bounds = self.current_bounds.copy()
        
        # Adaptive constraint minimums for B2 (updated every minute)
        self.current_constraint_mins = {
            'total_air_flow_min': 150.0,    # B2 specific constraint
            'wind_box_dp_min': 10.0,        # B2 specific constraint
            'steam_flow_min': 120.0,        # B2 specific constraint
            'coal_flow_min': 10.0,          # B2 specific constraint
            'aph_outlet_min': 190.0,        # B2 specific constraint
            'air_fuel_ratio_min': 2.11,     # Fixed minimum air/fuel ratio for safety
        }
        
        self.last_update = datetime.now()
        print(f"Adaptive bounds configured for {self.update_frequency} updates (every 1 minute) - Boiler B2")
        
    def add_real_time_data(self, parameter_values):
        'Add current parameter values to 1-minute rolling buffer.'
        for i, value in enumerate(parameter_values):
            if value is not None and isinstance(value, (int, float)) and not np.isnan(value):
                self.hourly_data[i].append(value)
    
    def calculate_adaptive_bounds_and_constraints(self):
        "Calculate new bounds based on past minute's data with balanced statistical approach for B2."
        new_bounds = []
        new_constraint_mins = self.current_constraint_mins.copy()
        
        for i in range(14):
            if len(self.hourly_data[i]) > self.min_data_points:
                data = list(self.hourly_data[i])
                
                # Use balanced percentiles for 1-minute updates
                p5 = np.percentile(data, 5)
                p95 = np.percentile(data, 95)
                range_buffer = (p95 - p5) * self.buffer_margin
                
                new_min = max(p5 - range_buffer, self.original_bounds[i][0] * 0.85)
                new_max = min(p95 + range_buffer, self.original_bounds[i][1] * 1.15)
                
                # Apply B2-specific constraint minimums
                if i == 1:  # Total Air Flow - B2 specific
                    new_min = max(new_min, 150.0)
                    new_constraint_mins['total_air_flow_min'] = new_min
                elif i == 2:  # Wind box DP - B2 specific
                    new_min = max(new_min, 10.0)
                    new_constraint_mins['wind_box_dp_min'] = new_min
                elif i == 6:  # Steam flow - B2 specific
                    new_min = max(new_min, 120.0)
                    new_constraint_mins['steam_flow_min'] = new_min
                elif i == 9:  # Coal flow - B2 specific
                    new_min = max(new_min, 10.0)
                    new_constraint_mins['coal_flow_min'] = new_min
                elif i == 10:  # APH outlet - B2 specific
                    new_min = max(new_min, 190.0)
                    new_constraint_mins['aph_outlet_min'] = new_min
                
                new_bounds.append((new_min, new_max))
            else:
                new_bounds.append(self.current_bounds[i])
                
        return new_bounds, new_constraint_mins
    
    def should_update_bounds(self):
        'Check if 1 minute has passed since last update.'
        return (datetime.now() - self.last_update) >= self.update_interval
    
    def update_bounds_and_constraints_if_needed(self):
        'Update field_0001 and field_0002 field_0003 field_0004 field_0012 field_0013.'
        if self.should_update_bounds():
            try:
                new_bounds, new_constraint_mins = self.calculate_adaptive_bounds_and_constraints()
                
                if self.validate_bounds_compatibility(new_bounds, new_constraint_mins):
                    self.current_bounds = new_bounds
                    self.current_constraint_mins = new_constraint_mins
                    self.last_update = datetime.now()
                    
                    # For 1-minute updates, keep some historical data for stability
                    for i in range(14):
                        if len(self.hourly_data[i]) > self.data_buffer_size * 0.8:
                            # Remove oldest 20% of data
                            for _ in range(int(len(self.hourly_data[i]) * 0.2)):
                                if self.hourly_data[i]:
                                    self.hourly_data[i].popleft()
                    
                    print(f"Adaptive bounds updated at {self.last_update.strftime('%H:%M:%S')} (1-minute frequency) - Boiler B2")
                    return True
                else:
                    print(f"Bounds validation failed at {datetime.now().strftime('%H:%M:%S')} - Boiler B2")
            except Exception as e:
                print(f"Bounds update error: {str(e)}")
        return False
    
    def validate_bounds_compatibility(self, new_bounds, new_constraint_mins):
        "Ensure new bounds don't create impossible constraint combinations for B2."
        try:
            for _ in range(3):
                test_values = [np.random.uniform(low, high) for (low, high) in new_bounds]
                
                # B2-specific constraint validation
                if test_values[1] < new_constraint_mins['total_air_flow_min']:
                    return False
                if test_values[2] < new_constraint_mins['wind_box_dp_min']:
                    return False
                if test_values[6] < new_constraint_mins['steam_flow_min']:
                    return False
                if test_values[9] < new_constraint_mins['coal_flow_min']:
                    return False
                if test_values[10] < new_constraint_mins['aph_outlet_min']:
                    return False
                if (test_values[1] / test_values[9]) < new_constraint_mins['air_fuel_ratio_min']:
                    return False
                    
            return True
        except:
            return False

    def update_constraints(self):
        'Update field_0005 field_0002 field_0006 field_0007 field_0008 field_0009 field_0010 field_0012 field_0013.'
        constraint_mins = self.bounds_manager.current_constraint_mins
        
        # Define adaptive constraints for B2 optimization
        self.cons = [
            # B2-specific adaptive constraints
            {'type': 'ineq', 'fun': lambda x: x[1] - constraint_mins['total_air_flow_min']},  # Total Air Flow ≥ adaptive_min
            {'type': 'ineq', 'fun': self.air_to_fuel_ratio},                                 # Air-to-fuel ratio ≥ 2.11
            {'type': 'ineq', 'fun': lambda x: x[9] - constraint_mins['coal_flow_min']},      # Coal Flow ≥ adaptive_min
            {'type': 'ineq', 'fun': lambda x: x[6] - constraint_mins['steam_flow_min']},     # Steam ≥ adaptive_min
            {'type': 'ineq', 'fun': lambda x: x[10] - constraint_mins['aph_outlet_min']},    # APH outlet ≥ adaptive_min
            {'type': 'ineq', 'fun': lambda x: x[2] - constraint_mins['wind_box_dp_min']},    # Wind box dp ≥ adaptive_min
            {'type': 'ineq', 'fun': self.scc_constraint},                                    # SCC ≤ max_scc
        ]

    def objective(self, x):
        input_df = pd.DataFrame([x], columns=self.model_features)
        return -self.model.predict(input_df)[0]

    def air_to_fuel_ratio(self, x):
        constraint_mins = self.bounds_manager.current_constraint_mins
        return (x[1] / x[9]) - constraint_mins['air_fuel_ratio_min']

    def scc_constraint(self, x):
        return self.max_scc - (x[9] / x[6])

    def validate_solution(self, x):
        return all(c['fun'](x) >= -1e-6 for c in self.cons)

    def multi_start_optimization(self, n_runs=50):
        'Multi-start optimization using current adaptive bounds for B2.'
        # Use current adaptive bounds for optimization
        current_bounds = self.bounds_manager.current_bounds
        
        best_solution = None
        best_efficiency = -np.inf
        
        for i in range(n_runs):
            # Generate random initial guess within current adaptive bounds
            x0 = [np.random.uniform(low, high) for (low, high) in current_bounds]
            try:
                res = minimize(
                    self.objective,
                    x0,
                    method='SLSQP',
                    bounds=current_bounds,  # Use adaptive bounds
                    constraints=self.cons   # Use adaptive constraints
                )
                print(f"[Run {i+1}] Success: {res.success}, Fun: {-res.fun if res.success else 'N/A'}, Message: {res.message}")
                if res.success and (-res.fun > best_efficiency):
                    best_solution = res.x
                    best_efficiency = -res.fun
            except Exception as e:
                print(f"[Run {i+1}] Optimization error: {e}")
                continue

        if best_solution is None:
            raise RuntimeError('All optimization attempts failed.')
        return best_solution, best_efficiency

    def safe_avg(self, values):
        numeric_values = [v for v in values if isinstance(v, (int, float))]
        if numeric_values:
            return sum(numeric_values) / len(numeric_values)
        else:
            return None

    def _get_batch_url(self, tag_url):
        parsed = urlparse(tag_url)
        return f"{parsed.scheme}://{parsed.netloc}/piwebapi/batch"

    def _batch_fetch(self, urls):
        if not urls:
            return

        batch_url = self._get_batch_url(urls[0])  # derive batch URL from first tag URL
        batch_payload = {
            f"req{i}": {
                'Method': 'GET',
                'Resource': url
            } for i, url in enumerate(urls)
        }

        try:
            response = requests.post(
                batch_url,
                json=batch_payload,
                auth=HTTPBasicAuth(self.username, self.password),
                verify=False
            )
            response.raise_for_status()
            batch_results = response.json()

            for i, url in enumerate(urls):
                result = batch_results.get(f"req{i}")
                value = None

                if result and 'Content' in result:
                    content = result['Content']
                    if isinstance(content, dict):
                        if 'Value' in content:
                            value = content['Value']
                        elif 'Value' in content.get('Value', {}):
                            value = content['Value']['Value']

                self._batch_cache[url] = value

        except Exception as e:
            print(f"Batch fetch error: {e}")
            for url in urls:
                self._batch_cache[url] = None

    def read_tag_value(self, url):
        with self._batch_lock:
            now = time.time()
            self._batch_urls.add(url)

            if (now - self._last_batch_time) > self._batch_interval:
                urls_to_fetch = list(self._batch_urls)
                self._batch_urls.clear()
                self._last_batch_time = now
                self._batch_fetch(urls_to_fetch)

            if url in self._batch_cache:
                return self._batch_cache[url]

        # fallback to single fetch if not cached yet
        try:
            response = requests.get(url, auth=HTTPBasicAuth(self.username, self.password), verify=False)
            response.raise_for_status()
            data = response.json()

            value = data.get('Value')
            if isinstance(value, (int, float)):
                return value
            if isinstance(value, dict):
                if 'Value' in value:
                    print(f"Extracted nested value: {value['Value']}")
                    return value['Value']
                else:
                    print(f"No numeric value found in nested dict: {value}")
                    return None

            print(f"Unexpected value format: {value}")
            return None

        except Exception as e:
            print(f"Error fetching from {url}: {e}")
            return None
