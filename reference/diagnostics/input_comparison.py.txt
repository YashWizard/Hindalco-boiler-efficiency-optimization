def fetch_pi_values(round_name):
    'Fetch current values from PI system with timestamps'
    print(f"\n{'='*80}")
    print(f"🔄 {round_name} - FETCHING PI TAG VALUES")
    print(f"⏰ Timestamp: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print(f"{'='*80}")
    
    values_dict = {}
    
    for tag_name, url in pi_tag_urls.items():
        try:
            # Fetch from PI Web API
            response = requests.get(
                url, 
                auth=HTTPBasicAuth(username, password), 
                verify=False, 
                timeout=10
            )
            response.raise_for_status()
            
            # Parse response
            data = response.json()
            value = data.get('Value')
            
            # Handle nested value structure
            if isinstance(value, dict) and 'Value' in value:
                value = value['Value']
            
            # Store and display
            values_dict[tag_name] = value
            print(f"{tag_name:30}: {value:>15.6f}")
            
        except requests.exceptions.RequestException as e:
            print(f"{tag_name:30}: ERROR - Network/HTTP: {e}")
            values_dict[tag_name] = None
        except ValueError as e:
            print(f"{tag_name:30}: ERROR - JSON Parse: {e}")
            values_dict[tag_name] = None
        except Exception as e:
            print(f"{tag_name:30}: ERROR - {type(e).__name__}: {e}")
            values_dict[tag_name] = None
    
    return values_dict

def compare_values(values1, values2):
    'Compare two sets of values to detect changes'
    print(f"\n{'='*80}")
    print('📊 REAL-TIME DATA VERIFICATION - CHANGE ANALYSIS')
    print(f"{'='*80}")
    
    changes_detected = 0
    total_valid_comparisons = 0
    
    print(f"{'Tag Name':<30} | {'Reading 1':>15} | {'Reading 2':>15} | {'Change':>15} | {'Status'}")
    print('-' * 95)
    
    for tag_name in pi_tag_urls.keys():
        val1 = values1.get(tag_name)
        val2 = values2.get(tag_name)
        
        if val1 is not None and val2 is not None:
            total_valid_comparisons += 1
            change = abs(val2 - val1)
            
            if change > 0.001:  # Threshold for detecting meaningful change
                changes_detected += 1
                status = '✅ CHANGED'
            else:
                status = '➡️ STABLE'
            
            print(f"{tag_name:<30} | {val1:>15.6f} | {val2:>15.6f} | {change:>15.6f} | {status}")
        else:
            print(f"{tag_name:<30} | {'ERROR':>15} | {'ERROR':>15} | {'N/A':>15} | ❌ ERROR")
    
    # Summary
    print('-' * 95)
    print(f"📈 SUMMARY:")
    print(f"   • Total Tags Compared: {total_valid_comparisons}")
    print(f"   • Tags with Changes: {changes_detected}")
    print(f"   • Change Detection Rate: {(changes_detected/max(1,total_valid_comparisons))*100:.1f}%")
    
    if changes_detected > 0:
        print(f"✅ REAL-TIME DATA CONFIRMED: {changes_detected} tags show live data changes!")
        print(f"   Your model IS reading real-time data from the PI system.")
    else:
        print(f"⚠️ NO CHANGES DETECTED: Data appears static.")
        print(f"   Check if plant is running or if there's a data delay.")
    
    return changes_detected > 0
