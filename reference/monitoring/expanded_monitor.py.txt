# Data storage
data_history = {tag: deque(maxlen=50) for tag in monitoring_tags.keys()}
time_history = deque(maxlen=50)

def fetch_tag_value(url):
    'Fetch single tag value from PI'
    try:
        response = requests.get(
            url,
            auth=HTTPBasicAuth(username, password),
            verify=False,
            timeout=10
        )
        response.raise_for_status()
        data = response.json()
        value = data.get('Value')
        if isinstance(value, dict) and 'Value' in value:
            value = value['Value']
        return float(value) if value is not None else None
    except Exception as e:
        print(f"Error fetching data: {e}")
        return None

def update_plot():
    'Create live updating plot with all parameters'
    # Calculate grid for all tags
    total_tags = len(monitoring_tags)
    cols = 8  # 8 columns for compact layout
    rows = (total_tags + cols - 1) // cols  # Ceiling division
    
    fig, axes = plt.subplots(rows, cols, figsize=(24, 4*rows))
    fig.suptitle('REAL-TIME PI DATA VERIFICATION - All Parameters', fontsize=16, fontweight='bold')
    
    # Handle different subplot configurations
    if rows == 1:
        axes = axes.reshape(1, -1) if total_tags > 1 else [axes]
    elif total_tags == 1:
        axes = [axes]
    
    # Flatten axes for easy iteration
    axes_flat = axes.flatten() if rows > 1 else axes
    
    # Plot each parameter
    for i, (tag_name, history) in enumerate(data_history.items()):
        if i < len(axes_flat) and len(history) > 0:
            ax = axes_flat[i]
            
            # Plot the data
            ax.plot(list(range(len(history))), list(history), 'o-', linewidth=2, markersize=3)
            
            # Shorter title for space
            short_name = tag_name.replace('process_signal_0002', '').replace('_', ' ')
            ax.set_title(short_name[:15], fontsize=9, fontweight='bold')
            
            # Labels
            ax.set_xlabel('Reading', fontsize=8)
            ax.set_ylabel('Value', fontsize=8)
            ax.grid(True, alpha=0.3)
            ax.tick_params(labelsize=7)
            
            # Add latest value and change indicator
            if len(history) > 1:
                latest = history[-1]
                previous = history[-2]
                change = latest - previous
                
                # Color code based on change
                color = 'green' if change > 0 else 'red' if change < 0 else 'blue'
                change_text = f"↑ +{change:.3f}" if change > 0 else f"↓ {change:.3f}" if change < 0 else '→ No Change'
                
                ax.text(0.02, 0.98, f'Latest: {latest:.3f}\n{change_text}',
                       transform=ax.transAxes, fontsize=8, fontweight='bold',
                       verticalalignment='top', color=color,
                       bbox=dict(boxstyle='round,pad=0.3', facecolor='yellow', alpha=0.7))
    
    # Hide unused subplots
    for i in range(total_tags, len(axes_flat)):
        axes_flat[i].set_visible(False)
    
    # Adjust spacing for compact layout
    plt.subplots_adjust(hspace=0.6, wspace=0.4, top=0.94, bottom=0.06)
    plt.show()

def print_data_status(cycle, values):
    'Print current data status'
    print(f"\nCYCLE {cycle} - {datetime.now().strftime('%H:%M:%S')}")
    print('-' * 60)
    
    # Show first 10 for console clarity, but monitor all in graphs
    count = 0
    for tag_name, value in values.items():
        if count >= 10:  # Limit console output
            break
        if value is not None:
            # Check if data changed
            if len(data_history[tag_name]) > 0:
                previous = data_history[tag_name][-1]
                change = value - previous
                status = '📈 CHANGED' if abs(change) > 0.001 else '➡️ STABLE'
                print(f"{tag_name:20}: {value:10.4f} | {status}")
            else:
                print(f"{tag_name:20}: {value:10.4f} | 🆕 FIRST")
        else:
            print(f"{tag_name:20}: {'ERROR':>10} | ❌ FAILED")
        count += 1
    
    if len(values) > 10:
        print(f"... and {len(values) - 10} more parameters (shown in graphs)")
